--- ex 2 - tables as computation representation
{
    if: (> x 1)
    them: (+ x 1)
    else: (- x 2)
}// now x = 3
(print x) -> prints 3
(eval {
	init: (define i 0)
	cond: (<= i 5)
	loop: (
		(print i)
		(increment i)
	)
}) -> prints 0 1 2 3 4 5
